/** Bandeja y revisión de aprobaciones desde memoria local. */ /** * Actualiza la bandeja y conserva la revisión solo mientras su versión siga vigente. */ function renderApprovals() { const list = state.data.approvals; $('#approval-count').textContent = list.length; $('#approval-list').innerHTML = list.length ? list.map(a=>`
Pendiente
`).join('') : '
No tienes aprobaciones pendientes.
'; if (state.selectedApproval) { const current=list.find(a=>a.aprobacion_id===state.selectedApproval.aprobacion_id); if (!current || current._version!==state.selectedApproval._version || current.compromiso._version!==state.selectedApproval.compromiso._version) { const id=state.selectedApproval.aprobacion_id; state.selectedApproval=null;$$('#approval-detail [data-action="decide"]').forEach(b=>b.disabled=true); if (!$('#approval-refresh-note')) $('#approval-detail').insertAdjacentHTML('afterbegin', `

La solicitud cambió.

`); } } else if (!$('#approval-detail textarea')) $('#approval-detail').textContent='Selecciona una aprobación para revisar.'; } /** * Abre una aprobación desde memoria local y fija las versiones que serán enviadas. * @param {*} id - Identificador del registro. * @param {string} draft - Comentario de revisión que se conservará al actualizar la aprobación. */ function openApproval(id, draft) { const a = state.data.approvals.find(x=>x.aprobacion_id===id); if (!a) return; const load = ++state.approvalLoad; state.selectedApproval = null; $('#approval-detail').textContent = 'Cargando solicitud…'; try { const detail = state.details[a.compromiso_id]; if (!detail) throw new Error('Actualiza los datos para revisar este compromiso.'); if (load !== state.approvalLoad) return; state.selectedApproval = a; const c = detail.commitment; const ev = detail.evidence.filter(e => e.aprobacion_id === a.aprobacion_id).slice(-1)[0]; $('#approval-detail').innerHTML = `

Revisión de cierre

${escapeHtml(a.aprobacion_id)}
Compromiso${escapeHtml(c.titulo)}
Responsable${escapeHtml(c.owner_name)}
Fecha objetivo${dateLabel(c.fecha_objetivo)}
Descripción${escapeHtml(c.descripcion || '—')}
Evidencia${ev?`${escapeHtml(ev.nombre_archivo || 'Abrir enlace')} ↗`:'Sin enlace'}
`; if (draft) $('#decision-comment').value = draft; } catch (error) { showToast(error.message); } }
